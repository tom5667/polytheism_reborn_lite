tenet_sacrificial_appeasement = {
	# name = {
	# 	first_valid = {
	# 		triggered_desc = {
	# 			trigger = { religion_tag = germanic_religion }
	# 			desc = tenet_gruesome_festivals_blot_name
	# 		}
	# 		desc = tenet_gruesome_festivals_name
	# 	}
	# }
	desc = tenet_sacrificial_appeasement_desc

	icon = tenet_gruesome_festivals

	piety_cost = {
		# If you've already got human sacrifice, this is a nice and easy jump.
		if = {
			limit = {
				rite_has_tenet = tenet_human_sacrifice
			}
			value = faith_tenet_cost_low
		}
		# Otherwise, follow the same pricing tiers as tenet_human_sacrifice.
		else_if = {
			limit = {
				rite_has_doctrine = pagan_hostility_doctrine
			}
			value = faith_tenet_cost_mid
		}
		else_if = {
			limit = {
				rite_religion_tag = hinduism_religion
			}
			value = faith_tenet_cost_high
		}
		else = {
			value = faith_tenet_cost_massive
		}

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_gruesome_festivals }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	is_shown = {
		has_fp1_dlc_trigger = yes
	}

	can_pick = {
		# No HumSac for Abrahamics: 'tis literally the religion-namer.
		NOR = {
			religion = religion:christianity_religion
			religion = religion:judaism_religion
			religion = religion:islam_religion
		}
		custom_description = {
			text = incompatible_tenet_pacifism_trigger
			NOT = {
				tenet:tenet_pacifism = { is_in_list = selected_doctrines }
			}
		}
		custom_description = {
			text = incompatible_tenet_dharmic_pacifism_trigger
			NOT = {
				tenet:tenet_dharmic_pacifism = { is_in_list = selected_doctrines }
			}
		}
		custom_description = {
			text = incompatible_tenet_human_sacrifice_trigger
			NOT = {
				tenet:tenet_human_sacrifice = { is_in_list = selected_doctrines }
			}
		}
		custom_description = {
			text = incompatible_tenet_sacrificial_ceremonies_trigger
			NOT = {
				tenet:tenet_sacrificial_ceremonies = { is_in_list = selected_doctrines }
			}
		}
	}

	parameters = {
		#human_sacrifice_active = yes
		gruesome_festivals_active
	}

	# 1.20 tenet virtues/sins. Like vanilla Gruesome Festivals (gregarious), plus the duty to appease the
	# gods: a willing victim as in the Roman devotio (Livy 8.9, Decius Mus).
	traits = {
		virtues = {
			gregarious = { weight = 15 scale = 1 }
			zealous = { weight = 10 scale = 1 }
		}
		sins = {
			cynical = { weight = 15 scale = 1 }
		}
	}
}
