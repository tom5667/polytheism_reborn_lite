# =============================================
# Polytheism Reborn - Court Positions
# File: pr_court_positions.txt
#
# 12 Flamen/Augur court positions for the Roman religion.
# Each Flamen is a priest dedicated to a specific deity.
# All share a learning-based aptitude structure but grant
# unique scaling modifiers reflecting their deity's domain.
# =============================================

# =============================================
# SHARED APTITUDE BLOCK (learning-based)
# =============================================
# All Flamen positions use the same aptitude calculation:
#   - Learning skill (x1.75, max 50)
#   - Mystic traits (mystic_1/2/3)
#   - Theologian trait
#   - Education learning traits
#   - Family business + low penalty
# Pater Mithras uses a prowess-based aptitude instead.

# =============================================
# COURT AUGUR
# =============================================
pr_court_augur_court_position = {
	sort_order = 250
	max_available_positions = 1
	skill = learning
	category = court_position_category_flamen

	opinion = {
		value = regular_court_position_opinion
	}

	aptitude_level_breakpoints = { 20 40 60 80 }

	aptitude = {
		value = 1
		add = {
			value = learning
			multiply = 1.75
			max = 50
			desc = court_position_skill_learning
		}
		if = {
			limit = { has_trait = mystic_3 }
			add = { value = 20 desc = court_position_mystic_3_trait }
		}
		else_if = {
			limit = { has_trait = mystic_2 }
			add = { value = 10 desc = court_position_mystic_2_trait }
		}
		else_if = {
			limit = { has_trait = mystic_1 }
			add = { value = 10 desc = court_position_mystic_1_trait }
		}
		else_if = {
			limit = { has_trait = theologian }
			add = { value = 10 desc = court_position_theologian_trait }
		}
		if = {
			limit = { has_trait = education_learning }
			if = {
				limit = { has_trait = education_learning_2 }
				add = { value = 8 desc = education_learning }
			}
			else_if = {
				limit = { has_trait = education_learning_3 }
				add = { value = 12 desc = education_learning }
			}
			else_if = {
				limit = { has_trait = education_learning_4 }
				add = { value = 16 desc = education_learning }
			}
			else_if = {
				limit = { has_trait = education_learning_5 }
				add = { value = 20 desc = education_learning }
			}
			else = {
				add = { value = 4 desc = education_learning }
			}
		}
		add = court_position_aptitude_family_business_value
		add = court_position_aptitude_low_penalty_value
		add = general_aptitude_bonus
	}

	is_shown = {
		has_religion = religion:roman_religion
		NOT = {
			OR = {
				government_has_flag = government_is_landless_adventurer
				government_has_flag = government_is_nomadic
			}
		}
	}

	is_shown_character = {
		scope:employee = {
			is_imprisoned = no
		}
	}

	valid_character = {
		scope:employee = {
			religion = religion:roman_religion
			OR = {
				has_trait = education_learning_1
				has_trait = education_learning_2
				has_trait = education_learning_3
				has_trait = education_learning_4
				has_trait = education_learning_5
				has_trait = mystic_1
				has_trait = mystic_2
				has_trait = mystic_3
				has_trait = theologian
				learning >= 11
			}
		}
	}

	revoke_cost = {
		prestige = {
			value = major_court_position_prestige_revoke_cost
			if = {
				limit = { exists = this is_ai = yes }
				multiply = 0
			}
			if = {
				limit = { exists = this has_character_flag = free_revoke_cost }
				multiply = 0
			}
		}
	}

	salary = {
		gold = {
			value = 0
			if = {
				limit = { scope:liege = { has_treasury = no } }
				add = medium_court_position_salary
			}
		}
		treasury = {
			value = 0
			if = {
				limit = { scope:liege = { has_treasury = yes } }
				add = medium_court_position_salary
			}
		}
		round = no
	}

	received_salary = {
		gold = {
			value = medium_court_position_salary
		}
		round = no
	}

	# Augur: divination and religious guidance
	scaling_employer_modifiers = {
		terrible = {
			monthly_court_grandeur_change_mult = 0.02
		}
		poor = {
			monthly_court_grandeur_change_mult = 0.05
		}
		average = {
			monthly_court_grandeur_change_mult = 0.10
		}
		good = {
			court_grandeur_baseline_add = 1
			monthly_court_grandeur_change_mult = 0.15
		}
		excellent = {
			court_grandeur_baseline_add = 2
			monthly_court_grandeur_change_mult = 0.25
		}
	}

	custom_employer_modifier_description = pr_court_augur_employer_custom_effect_description

	modifier = {
		learning = 1
	}

	on_court_position_received = {
		court_augur_title_accepted_effect = yes
	}
	on_court_position_revoked = {
		court_augur_title_revoked_effect = yes
	}
	on_court_position_invalidated = {
		court_augur_title_invalidated_effect = yes
	}
	on_court_position_vacated = {
		court_position_vacated_event_effect = {
			COURT_POSITION_TYPE = pr_court_augur_court_position
		}
	}

	ai_position_score = {
		value = 50
		add = court_position_debt_considerations_value
	}

	ai_candidate_score = {
		value = 50
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
	}
}
