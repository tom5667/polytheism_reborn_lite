# Roman Triggers
pr_augur_basic_requirements_trigger = {
	is_adult = yes
	is_imprisoned = no
	is_alive = yes # Because apparently there can be dead characters in the pool (remove when fixed)
	is_claimant = no # We don't want recruiting physicians to be a cheap way of getting claimants
	faith = root.faith
	has_trait = education_learning_1
	learning >= 13
}
pr_excellent_skill_augur_trigger = {
	pr_augur_basic_requirements_trigger = yes
	learning >= high_skill_rating
}
pr_high_skill_augur_trigger = {
	pr_augur_basic_requirements_trigger = yes
	OR = {
		#Empire/kingdom
		AND = {
			root = { highest_held_title_tier >= tier_kingdom }
			learning >= decent_skill_rating #Will be pushed up to high by adding physician traits
			NAND = {
				has_trait = lifestyle_mystic
				NOT = { has_trait = lifestyle_physician }
			}
		}
		#Duchy/county
		AND = {
			root = { highest_held_title_tier <= tier_duchy }
			NOR = {
				learning > high_skill_rating
				has_trait =  education_learning_3
				has_trait = lifestyle_mystic
			}
			learning >= mediocre_skill_rating #Will be pushed up to medium by adding physician traits
		}
	}
}
pr_low_skill_augur_trigger = {
	pr_augur_basic_requirements_trigger = yes
	NOR = {
		has_trait = lifestyle_mystic
		has_trait = education_learning_2
	}
	OR = {
		#Empire/kingdom
		AND = {
			root = { highest_held_title_tier >= tier_kingdom }
			learning < decent_skill_rating
			learning >= mediocre_skill_rating
		}
		#Duchy/county
		AND = {
			root = { highest_held_title_tier <= tier_duchy }
			learning < decent_skill_rating
			learning >= mediocre_skill_rating
		}
	}
}
pr_low_skill_augur_light_trigger = {
	pr_augur_basic_requirements_trigger = yes
	learning > mediocre_skill_rating
}
pr_mystic_augur_trigger = {
	pr_augur_basic_requirements_trigger = yes

	#For this one, we add mystic trait if they don't have it if their learning isn't awesome
	#Must be suitable mystic
	has_trait = lifestyle_mystic
	OR = {
		#Empire/kingdom
		AND = {
			root = { highest_held_title_tier >= tier_kingdom }
			learning >= decent_skill_rating #Will be pushed up to high by adding mystic traits
			NAND = {
				has_trait = education_learning_2
				NOT = { has_trait = lifestyle_mystic }
			}
		}
		#Duchy/county
		AND = {
			root = { highest_held_title_tier <= tier_duchy }
			NOR = {
				learning > high_skill_rating
				has_trait_xp = { trait = lifestyle_mystic value >= 100 }
				has_trait = education_learning_3
			}
			learning >= mediocre_skill_rating #Will be pushed up to medium by adding mystic traits
		}
	}
}

# Character scope: may this character be given a College seat? Used by the appoint interaction and by
# co-optation. An adult Roman of the gender the rite allows as clergy (as vanilla's appoint_cardinal_interaction,
# pam_interactions.txt:4697-4700), without a seat, and from a dynasty that holds no seat yet: no two members of one
# gens in the same college (Dio 39.17).
pr_college_can_receive_seat_trigger = {
	is_alive = yes
	faith = faith:roman_polytheism
	is_adult = yes
	# Men only (decided 2026-11-02): pontifices, rex sacrorum and flamines were men; women served as Vestals and as
	# flaminica beside a flamen. The Roman religion allows female clergy (doctrine_clerical_gender_either, for the
	# Vestals), so rite_has_allowed_gender_for_clergy below is not enough (in-game: three women co-opted).
	is_male = yes
	is_imprisoned = no
	is_incapable = no
	NOT = { exists = clerical_elector_title }
	faith = {
		NOT = { religious_head ?= prev }
	}
	save_temporary_scope_as = pr_college_candidate
	rite = { rite_has_allowed_gender_for_clergy = scope:pr_college_candidate }
	trigger_if = {
		limit = { exists = dynasty }
		faith = {
			NOT = {
				any_clerical_elector_title = {
					holder ?= { dynasty ?= scope:pr_college_candidate.dynasty }
				}
			}
		}
	}
	# No enemy of a sitting member: "ne cooptari quidem sacerdotem licebat qui cuiquam ex collegio esset inimicus"
	# (Cic. Fam. 3.10.9; Webb 2023, p.32). Decided 2026-10-18.
	faith = {
		NOT = {
			any_clerical_elector_title = {
				holder ?= {
					OR = {
						has_relation_rival = scope:pr_college_candidate
						has_relation_nemesis = scope:pr_college_candidate
					}
				}
			}
		}
	}
}

# Character or county scope: belongs to the Church of Rome. Before the Great Schism that is christian_faith with one of
# the rites vanilla moves to Catholicism (pam_effects.txt:7404-7487, 7556-7564); after it, faith:catholic.
pr_college_church_of_rome_trigger = {
	OR = {
		faith = faith:catholic
		AND = {
			faith = faith:christian_faith
			OR = {
				rite = rite:roman_rite
				rite = rite:insular_celtic
				rite = rite:mozarabic_church
				rite = rite:ambrosian_rite
				rite = rite:carolingian_christianity
				rite = rite:beneventan_rite
			}
		}
	}
}
