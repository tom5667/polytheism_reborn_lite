# Roman Triggers
pr_augur_basic_requirements_trigger = {
	is_adult = yes
	is_imprisoned = no
	is_alive = yes # Because apparently there can be dead characters in the pool (remove when fixed)
	is_claimant = no # We don't want recruiting physicians to be a cheap way of getting claimants
	faith = root.faith
	has_trait = education_learning_1
	learning >= 13
}
pr_excellent_skill_augur_trigger = {
	pr_augur_basic_requirements_trigger = yes
	learning >= high_skill_rating
}
pr_high_skill_augur_trigger = {
	pr_augur_basic_requirements_trigger = yes
	OR = {
		#Empire/kingdom
		AND = {
			root = { highest_held_title_tier >= tier_kingdom }
			learning >= decent_skill_rating #Will be pushed up to high by adding physician traits
			NAND = {
				has_trait = lifestyle_mystic
				NOT = { has_trait = lifestyle_physician }
			}
		}
		#Duchy/county
		AND = {
			root = { highest_held_title_tier <= tier_duchy }
			NOR = {
				learning > high_skill_rating
				has_trait =  education_learning_3
				has_trait = lifestyle_mystic
			}
			learning >= mediocre_skill_rating #Will be pushed up to medium by adding physician traits
		}
	}
}
pr_low_skill_augur_trigger = {
	pr_augur_basic_requirements_trigger = yes
	NOR = {
		has_trait = lifestyle_mystic
		has_trait = education_learning_2
	}
	OR = {
		#Empire/kingdom
		AND = {
			root = { highest_held_title_tier >= tier_kingdom }
			learning < decent_skill_rating
			learning >= mediocre_skill_rating
		}
		#Duchy/county
		AND = {
			root = { highest_held_title_tier <= tier_duchy }
			learning < decent_skill_rating
			learning >= mediocre_skill_rating
		}
	}
}
pr_low_skill_augur_light_trigger = {
	pr_augur_basic_requirements_trigger = yes
	learning > mediocre_skill_rating
}
pr_mystic_augur_trigger = {
	pr_augur_basic_requirements_trigger = yes

	#For this one, we add mystic trait if they don't have it if their learning isn't awesome
	#Must be suitable mystic
	has_trait = lifestyle_mystic
	OR = {
		#Empire/kingdom
		AND = {
			root = { highest_held_title_tier >= tier_kingdom }
			learning >= decent_skill_rating #Will be pushed up to high by adding mystic traits
			NAND = {
				has_trait = education_learning_2
				NOT = { has_trait = lifestyle_mystic }
			}
		}
		#Duchy/county
		AND = {
			root = { highest_held_title_tier <= tier_duchy }
			NOR = {
				learning > high_skill_rating
				has_trait_xp = { trait = lifestyle_mystic value >= 100 }
				has_trait = education_learning_3
			}
			learning >= mediocre_skill_rating #Will be pushed up to medium by adding mystic traits
		}
	}
}

# Cultus Deorum: the Roman religion without the Etruscan faith (issue #86, decided 2026-10-07).
# Use these instead of religion = religion:roman_religion for Roman (Cultus Deorum) content, so that
# reformed Roman faiths keep the content and the Etruscan faith (same religion) does not get it.
# A later Italic faith is excluded here too.
# Character scope.
pr_is_cultus_deorum_trigger = {
	religion = religion:roman_religion
	NOT = { faith = faith:etruscan_polytheism }
}
# Faith scope.
pr_is_cultus_deorum_faith_trigger = {
	religion = religion:roman_religion
	NOT = { this = faith:etruscan_polytheism }
}

# Override of the vanilla 1.20 trigger (common/scripted_triggers/00_major_decision_triggers.txt) with one PR line,
# so restore_roman_empire_decision itself stays vanilla. Re-copy on game updates (#167).
restore_roman_empire_decision_religion_culture_trigger = {
	custom_tooltip = {
		text = restore_roman_empire_decision_religion_culture_tt
		OR = {
			religion = {
				OR = {
					is_in_family = rf_abrahamic
					this = religion:hellenism_religion
				}
			}
			pr_is_cultus_deorum_trigger = yes # PR: Roman polytheists may restore Rome (#167)
			culture = {
				OR = {
					has_cultural_pillar = heritage_latin
					has_cultural_pillar = heritage_byzantine
				}
			}
		}
	}
}
