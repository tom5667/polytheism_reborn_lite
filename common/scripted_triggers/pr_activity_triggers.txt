pr_feast_is_valid_trigger = {
		scope:host = {
			is_alive = yes
			is_capable_adult = yes
			is_imprisoned = no
			is_landed = yes
			NOT = { is_incapable = yes }
		}
		# If no one shows up
		trigger_if = {
			limit = {
				is_current_phase_active = yes
			}
			has_attending_activity_guests = yes
		}
}

pr_feast_can_be_activity_guest_trigger = {
		faith.religion = scope:host.faith.religion
		is_adult = yes
		is_healthy = yes
		in_diplomatic_range = scope:host
}
