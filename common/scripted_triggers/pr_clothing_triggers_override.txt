# PR (#76): holders of a Roman College seat are not cardinals. Without this every seat holder wears the cardinal's hat,
# robe and cloak and gets the cardinal animations (03_headgear_religious, 03_clothes_religious, 06_cloaks_special,
# animations.modifierpack). Decided 2026-10-21.
portrait_is_cardinal_trigger = {
	exists = this
	has_pam_dlc_trigger = yes
	exists = clerical_elector_title
	portrait_religious_head_no_poverty_trigger = no
	NOT = {
		rite = {
			rite_has_parameter = vows_of_poverty_active
		}
	}
	NOT = { religion = religion:roman_religion } # PR (#76)
}
