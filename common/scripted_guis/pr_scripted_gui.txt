pr_restore_cult = {
	scope = character
	is_shown = {}
	effect = {
		if = {
			limit = {
				OR = {
					religion = religion:roman_religion
					faith = faith:kemetic_polytheism
					religion = religion:hellenism_religion
				}
			}
			trigger_event = pr_roman_cult.0042
		}
	}
}

pr_get_cultists_list = {
	scope = character
	is_shown = {}
	effect = {
		root = {
			clear_variable_list = pr_cultists_list
		}
		root = {
			every_living_character = {
				limit = {
					OR = {
						AND = {
							root = { has_trait = mithras_cult }
							has_trait = mithras_cult
							religion = religion:roman_religion
						}
						AND = {
							root = { has_trait = bacchus_cult }
							has_trait = bacchus_cult
							religion = religion:roman_religion
						}
						AND = {
							root = { has_trait = isis_cult }
							has_trait = isis_cult
							OR = {
								religion = religion:roman_religion
								religion = religion:hellenism_religion
								faith = faith:kemetic_polytheism
							}
						}
						AND = {
							root = { has_trait = sol_invictus_cult }
							has_trait = sol_invictus_cult
							religion = religion:roman_religion
						}
						AND = {

						AND = {

							has_trait = eleusinian_cult
							OR = {
								religion = religion:roman_religion
								religion = religion:hellenism_religion
