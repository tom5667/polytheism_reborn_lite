# Root = none (empty scope)
on_game_start = {
	on_actions = {
		PR_check
		TFE_check
		RICE_check
		PR_BM_700_check
	}
}

# Root = none (empty scope)
on_game_start_after_lobby = {
	on_actions = {
		polytheism_reborn_on_game_start
	}
}

# --- Mod Load Checks ---

PR_check = {
	effect = {
		set_global_variable = {
			name = PR_is_loaded
			value = yes
		}
	}
}

# Bookmark 700 AD check
PR_BM_700_check = {
	effect = {
		if = {
			limit = {
				current_date < 865.1.1
			}
			set_global_variable = {
				name = PR_BM_700_is_loaded
				value = yes
			}
		}
	}
}

# The Fallen Empire mod check
TFE_check = {
	effect = {
		if = {
			limit = {
				current_date < 699.1.1
			}
			set_global_variable = {
				name = TFE_is_loaded
				value = yes
			}
		}
	}
}

# RICE mod check
RICE_check = {
	effect = {
		if = {
			limit = {
				OR = {
					exists = global_var:is_RICE_loaded_global_variable
					exists = religion:inuit_religion
				}
			}
			pr_rice_culture_tradition_effect = yes
		}
	}
}

# --- Game Start Events ---

polytheism_reborn_on_game_start = {
	effect = {
		# Emperor Julian nickname choice
		if = {

				game_start_date = 361.11.17
			}
			title:h_roman_empire.holder ?= {
				trigger_event = pr_roman.0031
			}
		}
	}
}

# --- Temple of Vesta ---

# Root = character who holds Roma
PR_check_temple_of_vesta_owner_roma_yearly_pulse = {
	trigger = {
		has_title = title:c_roma
		title:c_roma = { has_county_modifier = open_temple_of_vesta }
		NOR = {
			has_character_modifier = burning_flame_of_vesta
		}
	}
	effect = {
		add_character_modifier = burning_flame_of_vesta
	}
}

# Root = character who lost Roma or changed religion
PR_check_death_or_lost_roma_temple_of_vesta_yearly_pulse = {
	trigger = {
		OR = {
			AND = {
				has_character_modifier = burning_flame_of_vesta
				NOT = {
					has_title = title:c_roma
				}
			}
			AND = {
				has_character_modifier = burning_flame_of_vesta
				NOT = {
					pr_is_cultus_deorum_trigger = yes
				}
			}
		}
	}
	effect = {
		remove_character_modifier = burning_flame_of_vesta
	}
}

# Root = Celtic-religion player with vassals while a Chief Druid lives (#146).
# Players only: the leader check scans every living character.
PR_celtic_druid_random_events_yearly_pulse = {
	trigger = {
		is_ai = no
		religion = religion:celtic_religion
		any_vassal = { count >= 2 }
		any_living_character = { has_trait = druidic_circle_leader }
	}
	random_events = {
		chance_to_happen = 10
		100 = pr_celtic.0421 # the druid judges a dispute between two vassals
		100 = pr_celtic.0423 # the druid as teacher: the fading verse
	}
}

# --- Patron-god traits -> modifiers (2026-10-10) ---
# Old saves: a patron trait still held becomes the modifier with the same key.
PR_convert_patron_traits_yearly_pulse = {
	trigger = {
		OR = {
			has_trait = pr_cdr_iupiter
			has_trait = pr_cdr_iuno
			has_trait = pr_cdr_mars
			has_trait = pr_cdr_venus
			has_trait = pr_cdr_lares
			has_trait = pr_cdr_mercurius
			has_trait = pr_cdr_minerva
			has_trait = pr_cdr_apollo
			has_trait = pr_cdr_ceres
			has_trait = pr_mithras_mithras
			has_trait = pr_mithras_arimanius
			has_trait = pr_mithras_oromazes
			has_trait = pr_hellenic_zeus
			has_trait = pr_hellenic_hera
			has_trait = pr_hellenic_ares
			has_trait = pr_hellenic_aphrodite
			has_trait = pr_hellenic_hermes
			has_trait = pr_hellenic_athena
			has_trait = pr_hellenic_demeter
			has_trait = pr_kemetic_amun_ra
			has_trait = pr_kemetic_ra
			has_trait = pr_kemetic_isis
			has_trait = pr_kemetic_montu
			has_trait = pr_kemetic_hathor
		}
	}
	effect = {
		if = {
			limit = { has_trait = pr_cdr_iupiter }
			remove_trait = pr_cdr_iupiter
			add_character_modifier = pr_cdr_iupiter
		}
		if = {
			limit = { has_trait = pr_cdr_iuno }
			remove_trait = pr_cdr_iuno
			add_character_modifier = pr_cdr_iuno
		}
		if = {
			limit = { has_trait = pr_cdr_mars }
			remove_trait = pr_cdr_mars
			add_character_modifier = pr_cdr_mars
		}
