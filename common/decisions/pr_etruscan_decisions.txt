# --- Create Etruria Decision ---
pr_etruscan_create_etruria_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/pr_etruscan.dds"
    }
    desc = pr_etruscan_create_etruria_decision_desc
    decision_tooltip = pr_etruscan_create_etruria_decision_tooltip
    decision_group_type = major

    ai_check_interval = 4

    is_shown = {
        is_ruler = yes
        is_landed = yes
        OR = {
            has_game_rule = pr_ancient_kingdoms_decisions_on
            AND = {
                has_game_rule = pr_ancient_kingdoms_decisions_player
                is_ai = no
            }
        }
        OR = {
            religion = religion:etruscan_religion
            culture = culture:pr_etruscan
        }
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:pr_etruscan_create_etruria_decision
            }
        }
    }

    is_valid = {
        has_title = title:d_pisa
        has_title = title:d_toscana
        OR = {
            religion = religion:etruscan_religion
            culture = culture:pr_etruscan
        }
        is_imprisoned = no
        is_at_war = no
    }

    cost = {
        gold = pr_kingdom_restore_gold_cost
        prestige = pr_kingdom_restore_prestige_cost
    }

    effect = {
        add_to_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:pr_etruscan_create_etruria_decision
        }
        custom_tooltip = pr_etruscan_create_etruria_decision_effect_tooltip
        title:d_pisa = { save_scope_as = pisa }
        title:d_toscana = { save_scope_as = toscana }

        create_title_and_vassal_change = {
            type = created
            save_scope_as = title_change
            add_claim_on_loss = yes
        }
        title:k_etruria = {
            change_title_holder = {
                holder = root
                change = scope:title_change
            }
        }
        resolve_title_and_vassal_change = scope:title_change
        title:d_toscana = { set_de_jure_liege_title = title:k_etruria }
        title:d_pisa = { set_de_jure_liege_title = title:k_etruria }
    }

    ai_potential = {
        is_ruler = yes
        has_title = title:d_pisa
        has_title = title:d_toscana
    }

    ai_will_do = {
        base = 100
        modifier = { add = 20 gold >= pr_kingdom_restore_gold_cost }
        modifier = { add = 20 prestige >= pr_kingdom_restore_prestige_cost }
    }
}
