# --- Return to Polytheism Decision ---
pr_return_polytheism_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/room_with_scrolls.tga"
	}
	decision_group_type = polytheism
	desc = pr_return_polytheism_decision_desc
	decision_tooltip = pr_return_polytheism_decision_tooltip

	is_shown = {
		AND = {
			OR = {
				learning >= 1
				has_trait = cynical
			}
			NAND = {
				is_ai = no
				is_ai = no
				OR = {
					has_character_flag = converted_pr_religion_this_lifetime
				}
			}
			NOT = {
				any_secret = {
					type = secret_crypto_religionist
				}
			}
			NOT = {
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:pr_return_polytheism_decision
				}
			}
		}
	}

	cooldown = { years = 2 }

	is_valid = {
		always = yes
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes

		custom_description = {
			text = can_only_change_religion_once
			NOT = { has_character_flag = converted_pr_religion_this_lifetime }
		}
		is_imprisoned = no
		is_at_war = no
		NOT = {
			has_trait = zealous
		}
	}

	effect = {
		trigger_event = pr.0018
		save_scope_as = actor
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:pr_return_polytheism_decision
			}
		}
	}
}
