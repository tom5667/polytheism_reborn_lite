# --- Embrace Babylonian Culture Decision ---
pr_mesopotamian_embrace_babylonian_culture_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/pr_mesopotamian.dds"
    }
    desc = pr_mesopotamian_embrace_babylonian_culture_decision_desc
    decision_tooltip = pr_mesopotamian_embrace_babylonian_culture_decision_tooltip
    decision_group_type = polytheism

    is_shown = {
        is_ai = no
        is_landed = yes
        OR = {
            faith = faith:babylonian_polytheism
            faith = { has_doctrine_parameter = special_doctrine_babylonian_patron_god_active }
        }
        culture = { has_cultural_pillar = heritage_pr_mesopotamian }
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_restored_babylonian_egc
            }
        }
    }

    is_valid = {
        is_adult = yes
        is_imprisoned = no
        completely_controls = title:c_baghdad
    }

    is_valid_showing_failures_only = {
        is_capable_adult = yes
        is_imprisoned = no
        is_at_war = no
        is_independent_ruler = yes
    }

    cost = {
        prestige = 1000
    }

    effect = {
        trigger_event = pr_mesopotamian.0001
        save_scope_as = actor
        hidden_effect = {
            add_to_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_restored_babylonian_egc
            }
        }
    }

    ai_check_interval = 90

    ai_potential = {
        always = yes
    }

    ai_will_do = {
        base = 100
    }
}

# --- Create Kingdom of Babylon Decision ---
pr_mesopotamian_kingdom_of_babylon_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/pr_mesopotamian.dds"
    }
    desc = pr_mesopotamian_kingdom_of_babylon_decision_desc
    decision_tooltip = pr_mesopotamian_kingdom_of_babylon_decision_tooltip
    decision_group_type = major

    ai_check_interval = 4

    is_shown = {
        is_ruler = yes
        is_landed = yes
        OR = {
            has_game_rule = pr_ancient_kingdoms_decisions_on
            AND = {
                has_game_rule = pr_ancient_kingdoms_decisions_player
                is_ai = no
            }
        }
        OR = {
            culture = culture:pr_babylonian
            religion = religion:mesopotamian_religion
        }
        is_imprisoned = no
        is_at_war = no
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:pr_mesopotamian_kingdom_of_babylon_decision
            }
        }
    }

    is_valid = {
        completely_controls = title:d_kufa
        completely_controls = title:d_basra
        completely_controls = title:d_samarra
        completely_controls = title:d_wasit
        completely_controls = title:d_baghdad
        OR = {
            culture = culture:pr_babylonian
            religion = religion:mesopotamian_religion
        }
    }

    cost = {
        gold = pr_kingdom_restore_gold_cost
        prestige = pr_kingdom_restore_prestige_cost
    }

    effect = {
        add_to_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:pr_mesopotamian_kingdom_of_babylon_decision
        }
        custom_tooltip = pr_mesopotamian_kingdom_of_babylon_decision_effect_tooltip
        title:d_kufa = { save_scope_as = kufa }
        title:d_basra = { save_scope_as = basra }
        title:d_samarra = { save_scope_as = samarra }
        title:d_wasit = { save_scope_as = wasit }
        title:d_baghdad = { save_scope_as = baghdad }

        create_title_and_vassal_change = {
            type = created
            save_scope_as = title_change
            add_claim_on_loss = yes
        }
        title:k_babylon = {
            change_title_holder = {
                holder = root
                change = scope:title_change
            }
        }
        resolve_title_and_vassal_change = scope:title_change
        title:d_kufa = { set_de_jure_liege_title = title:k_babylon }
        title:d_basra = { set_de_jure_liege_title = title:k_babylon }
        title:d_samarra = { set_de_jure_liege_title = title:k_babylon }
        title:d_wasit = { set_de_jure_liege_title = title:k_babylon }
        title:d_baghdad = { set_de_jure_liege_title = title:k_babylon }
        hidden_effect = {
            title:k_babylon = { set_de_jure_liege_title = title:e_mesopotamia }
        }
    }

    ai_potential = {
        is_ruler = yes
        completely_controls = title:d_baghdad
        completely_controls = title:d_kufa
    }

    ai_will_do = {
        base = 100
        modifier = { add = 20 gold >= pr_kingdom_restore_gold_cost }
        modifier = { add = 20 prestige >= pr_kingdom_restore_prestige_cost }
    }
}

    ai_check_interval = 4

    is_shown = {
        is_ruler = yes
        is_landed = yes
        OR = {
            has_game_rule = pr_ancient_kingdoms_decisions_on
            AND = {
                has_game_rule = pr_ancient_kingdoms_decisions_player
                is_ai = no
            }
        }
        OR = {
            religion = religion:mesopotamian_religion
            culture = culture:pr_assyrian
            culture = culture:pr_sumerian
            culture = culture:pr_babylonian
        }
        is_imprisoned = no
        is_at_war = no
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:pr_mesopotamian_restore_empire_decision
            }
        }
    }

    is_valid = {
        has_title = title:k_babylon
        has_title = title:k_sumer
        has_title = title:k_assyria
        OR = {
            religion = religion:mesopotamian_religion
            culture = culture:pr_assyrian
            culture = culture:pr_sumerian
            culture = culture:pr_babylonian

        is_imprisoned = no
        is_at_war = no

        add_to_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:pr_mesopotamian_restore_empire_decision
        }
        custom_tooltip = pr_mesopotamian_restore_empire_decision_effect_tooltip
        title:k_babylon = { save_scope_as = kingdom_babylon }
        title:k_sumer = { save_scope_as = kingdom_sumer }
