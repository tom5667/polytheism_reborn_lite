pr_celtic_restore_high_priesthood_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/celtic_druid.dds"
    }
    desc = pr_celtic_restore_high_priesthood_decision_desc
    decision_tooltip = pr_celtic_restore_high_priesthood_decision_tooltip
    decision_group_type = polytheism

    ai_check_interval = 120

    is_shown = {
        is_ruler = yes
        is_landed = yes
        NOR = {  #Can be redone if somehow the High Priest title is destroyed
            exists = faith.religious_head
            faith = { has_doctrine = doctrine_no_head } #Only if Faith is meant to have one.
        }
        OR = {
            faith = faith:gaelic_polytheism
            faith = faith:brythonic_polytheism
            faith = faith:celtic_iberian_polytheism
            faith = faith:gaulish_polytheism
        }
    }

    is_valid = {
        piety_level >= high_piety_level
        custom_description = {
            text = restore_high_priesthood_holy_sites
            faith = {
                any_holy_site = {
                    OR = {
                        county.holder = root
                        county.holder.top_liege = { this = root }
                    }
                    count >= 3
                }
            }
        }
    }

    is_valid_showing_failures_only = {
        is_capable_adult = yes
        is_imprisoned = no
        is_independent_ruler = yes
        is_at_war = no
    }

    cost = {
        piety = pr_high_priesthood_piety_cost
    }

    effect = {
        custom_tooltip = pr_celtic_restore_high_priesthood_decision_effect_tooltip
        trigger_event = {
            id = pr_celtic.0001
        }
    }

    ai_potential = {
        is_ruler = yes
        is_landed = yes
        piety_level >= high_piety_level
    }

    ai_will_do = {
        base = 100
        modifier = { add = 20 piety >= pr_high_priesthood_piety_cost }
    }
}

pr_celtic_create_empire_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/celtic_fort.dds"
    }
    desc = pr_celtic_create_empire_decision_desc
    decision_tooltip = pr_celtic_create_empire_decision_tooltip
    decision_group_type = polytheism

    ai_check_interval = 4

    is_shown = {
        is_ruler = yes
        is_landed = yes
        OR = {
            has_game_rule = pr_ancient_kingdoms_decisions_on
            AND = {
                has_game_rule = pr_ancient_kingdoms_decisions_player
                is_ai = no
            }
        }

        AND = {
            has_title = title:k_high_kingdom_of_ireland
            has_title = title:k_scotland
            has_title = title:k_wales
            has_title = title:k_england
        }
        is_imprisoned = no
        is_at_war = no
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:pr_celtic_create_empire_decision
            }
        }
    }

    is_valid = {
        has_title = title:k_high_kingdom_of_ireland
        has_title = title:k_scotland
        has_title = title:k_wales
        has_title = title:k_england
    }

    cost = {
        gold = pr_empire_restore_gold_cost
        prestige = pr_empire_restore_prestige_cost
    }

    effect = {
        add_to_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:pr_celtic_create_empire_decision
        }
        custom_tooltip = pr_celtic_create_empire_decision_effect_tooltip
        title:k_high_kingdom_of_ireland = { save_scope_as = high_kingdom_ireland }
        title:k_scotland = { save_scope_as = kingdom_scotland}
        title:k_wales = { save_scope_as = kingdom_wales }
        title:k_england = { save_scope_as = kingdom_england }

        create_title_and_vassal_change = {
            type = created
            save_scope_as = title_change
            add_claim_on_loss = yes
        }
        title:e_celtic_empire ={
            change_title_holder ={
                holder = root
                change = scope:title_change
            }
        }
        resolve_title_and_vassal_change = scope:title_change

        # title:k_high_kingdom_of_ireland = { set_de_jure_liege_title = title:e_celtic_empire }
        # title:k_scotland = { set_de_jure_liege_title = title:e_celtic_empire }
        # title:k_wales = { set_de_jure_liege_title = title:e_celtic_empire }
        # title:k_england = { set_de_jure_liege_title = title:e_celtic_empire }
        hidden_effect = {
            trigger_event = pr_celtic.0403
        }
    }

    ai_potential = {
        is_ruler = yes
        is_landed = yes
        has_title = title:k_high_kingdom_of_ireland
        has_title = title:k_scotland
