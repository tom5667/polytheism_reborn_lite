# --- Choose Main Deity (Greek) Decision ---

# --- Embrace Hellen Culture Decision ---
pr_greek_hellen_culture_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/greek_temple.dds"
    }
    desc = pr_greek_hellen_culture_decision_desc
    decision_tooltip = pr_greek_hellen_culture_decision_tooltip
    decision_group_type = polytheism

    is_shown = {
        is_ai = no
        is_landed = yes
        is_adult = yes
        is_imprisoned = no
        faith = faith:hellenic_pagan
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_restored_hellen_culture
            }
        }
    }

    is_valid = {
        completely_controls = title:c_attica
        culture = culture:greek
    }

    is_valid_showing_failures_only = {
        is_capable_adult = yes
        is_imprisoned = no
        is_at_war = no
    }

    cost = {
        prestige = 1000
    }

    effect = {
        trigger_event = pr_greek_decision.0004
        save_scope_as = actor
        hidden_effect = {
            add_to_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_restored_hellen_culture
            }
        }
    }

    ai_check_interval = 90

    ai_potential = {
        always = yes
    }

    ai_will_do = {
        base = 100
    }
}

# --- Create Pergamon Kingdom Decision ---
pr_greek_pergamon_kingdom_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/greek_temple.dds"
    }
    desc = pr_greek_pergamon_kingdom_decision_desc
    decision_tooltip = pr_greek_pergamon_kingdom_decision_tooltip
    decision_group_type = major

    ai_check_interval = 4

    is_shown = {
        is_ruler = yes
        is_landed = yes
        OR = {
            has_game_rule = pr_ancient_kingdoms_decisions_on
            AND = {
                has_game_rule = pr_ancient_kingdoms_decisions_player
                is_ai = no
            }
        }
        OR = {
            religion = religion:roman_religion
            religion = religion:hellenism_religion
            culture = culture:greek
            culture = culture:pr_greek
            culture = culture:roman
        }
        is_imprisoned = no
        is_at_war = no
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_recreated_pergamon
            }
        }
    }

    is_valid = {
        completely_controls = title:c_abydos
        completely_controls = title:c_aeolis
        completely_controls = title:d_thracesia
        completely_controls = title:c_mysia
        completely_controls = title:c_pacatiana
        OR = {
            culture = culture:greek
            culture = culture:pr_greek
            culture = culture:roman
        }
        OR = {
            religion = religion:roman_religion
            religion = religion:hellenism_religion
        }
    }

    cost = {
        gold = pr_kingdom_restore_gold_cost
        prestige = pr_kingdom_restore_prestige_cost
    }

    effect = {
        add_to_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:flag_recreated_pergamon
        }
        custom_tooltip = pr_greek_pergamon_kingdom_decision_effect_tooltip

        title:d_aegean_islands = { save_scope_as = aegean }
        title:d_ephese = { save_scope_as = ephese }
        title:d_thracesia = { save_scope_as = thracesia }
        title:d_opsikion = { save_scope_as = opsikion }
        title:d_anatolia = { save_scope_as = anatolia }

        create_title_and_vassal_change = {
            type = created
            save_scope_as = title_change
            add_claim_on_loss = yes
        }
        title:k_pergamon = {
            change_title_holder = {
                holder = root
                change = scope:title_change
            }
        }
        resolve_title_and_vassal_change = scope:title_change
        title:d_aegean_islands = { set_de_jure_liege_title = title:k_pergamon }
        title:d_ephese = { set_de_jure_liege_title = title:k_pergamon }
        title:d_thracesia = { set_de_jure_liege_title = title:k_pergamon }
        title:d_opsikion = { set_de_jure_liege_title = title:k_pergamon }
        title:d_anatolia = { set_de_jure_liege_title = title:k_pergamon }
    }

    ai_potential = {
        is_ruler = yes
        completely_controls = title:c_aeolis
        completely_controls = title:d_thracesia
    }

    ai_will_do = {
        base = 100
        modifier = { add = 20 gold >= pr_kingdom_restore_gold_cost }
        modifier = { add = 20 prestige >= pr_kingdom_restore_prestige_cost }
    }
}

# --- Set Capital Pergamon Decision ---
pr_greek_set_capital_pergamon_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/greek_temple.dds"
    }
    desc = pr_greek_set_capital_pergamon_decision_desc
    decision_tooltip = pr_greek_set_capital_pergamon_decision_tooltip
    decision_group_type = major

    ai_check_interval = 12

    is_shown = {
        is_ruler = yes
        is_landed = yes
        has_title = title:k_pergamon
        NOT = { capital_county = { this = title:c_aeolis } }
        OR = {
            title:c_aeolis.holder = {
                any_liege_or_above = { this = root }
            }
            title:c_aeolis.holder = { this = root }
        }
    }

    is_valid = {
    }

    is_valid_showing_failures_only = {
        is_capable_adult = yes
        is_imprisoned = no
        is_independent_ruler = yes
    }

    effect = {
        if = {
            limit = {
                NOT = { title:c_aeolis.holder = { this = root } }
            }
            create_title_and_vassal_change = {
                type = returned
                save_scope_as = change
                add_claim_on_loss = no
            }
            title:c_aeolis = {
                change_title_holder = {
                    holder = root
                    change = scope:change
                }
            }
            resolve_title_and_vassal_change = scope:change
        }
        set_realm_capital = title:c_aeolis
    }

    ai_potential = {
        has_title = title:k_pergamon
    }

    ai_will_do = {
        base = 100
    }
}
