# --- Create Kingdom of Illyria Decision ---
pr_illyrian_kingdom_of_illyria_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/pr_dacian.dds"
    }
    desc = pr_illyrian_kingdom_of_illyria_decision_desc
    decision_tooltip = pr_illyrian_kingdom_of_illyria_decision_tooltip
    decision_group_type = major

    ai_check_interval = 4

    is_shown = {
        is_ruler = yes
        is_landed = yes
        OR = {
            has_game_rule = pr_ancient_kingdoms_decisions_on
            AND = {
                has_game_rule = pr_ancient_kingdoms_decisions_player
                is_ai = no
            }
        }
        religion = religion:illyrian_religion
        is_imprisoned = no
        is_at_war = no
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:pr_illyrian_kingdom_of_illyria_decision
            }
        }
    }

    is_valid = {
        completely_controls = title:d_dalmatia
        completely_controls = title:d_dyrrachion
        completely_controls = title:d_duklja
        completely_controls = title:d_usora
        religion = religion:illyrian_religion
    }

    cost = {
        gold = pr_kingdom_restore_gold_cost
        prestige = pr_kingdom_restore_prestige_cost
    }

    effect = {
        add_to_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:pr_illyrian_kingdom_of_illyria_decision
        }
        custom_tooltip = pr_illyrian_kingdom_of_illyria_decision_effect_tooltip
        title:d_dalmatia = { save_scope_as = dalmatia }
        title:d_dyrrachion = { save_scope_as = dyrrachion }
        title:d_duklja = { save_scope_as = duklja }
        title:d_usora = { save_scope_as = usora }

        create_title_and_vassal_change = {
            type = created
            save_scope_as = title_change
            add_claim_on_loss = yes
        }
        title:k_pr_illyria = {
            change_title_holder = {
                holder = root
                change = scope:title_change
            }
        }
        resolve_title_and_vassal_change = scope:title_change
        title:d_dalmatia = { set_de_jure_liege_title = title:k_pr_illyria }
        title:d_dyrrachion = { set_de_jure_liege_title = title:k_pr_illyria }
        title:d_duklja = { set_de_jure_liege_title = title:k_pr_illyria }
        title:d_usora = { set_de_jure_liege_title = title:k_pr_illyria }
    }

    ai_potential = {
        is_ruler = yes
        completely_controls = title:d_dalmatia
        completely_controls = title:d_dyrrachion
    }

    ai_will_do = {
        base = 100
        modifier = { add = 20 gold >= pr_kingdom_restore_gold_cost }
        modifier = { add = 20 prestige >= pr_kingdom_restore_prestige_cost }
    }
}

            }
        }
    }

    ai_check_interval = 90

    ai_potential = {
        always = yes
    }

    ai_will_do = {
        base = 100
    }
}
