restore_carthage_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/carthago.dds"
    }
    decision_group_type = major

    ai_check_interval_by_tier = {
        barony = 0
        county = 0
        duchy = 0
        kingdom = 120
        empire = 0
        hegemony = 0
    }

    desc = restore_carthage_decision_desc
    selection_tooltip = restore_carthage_decision_tooltip

    is_shown = {
        is_ruler = yes
        is_playable_character = yes

        OR = {
            culture = { has_cultural_pillar = heritage_syriac }
            culture = culture:levantine
            culture = culture:pr_punic
        }

        # Not repeatable.
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_restored_carthage
            }
        }
    }

    is_valid = {
        is_independent_ruler = yes
        highest_held_title_tier < tier_empire

        prestige_level >= 4

        completely_controls_region = custom_carthage
        completely_controls_region = custom_tripolitania
        completely_controls_region = custom_mauretania
        completely_controls_region = custom_carthago_nova

    }

    is_valid_showing_failures_only = {
        top_liege = this
        is_available_adult = yes
    }

    cost = {
        gold = {
            value = 1200
        }
    }

    effect = {
        save_scope_as = founder
        show_as_tooltip = { restore_carthage_scripted_effect = yes }

        hidden_effect = {
            add_to_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:flag_restored_carthage
            }
        }

        trigger_event = major_decisions.3300
    }
}
pr_canaanite_choose_main_deity_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/fp1_decisons_hold_bolt.dds"
    }
    decision_group_type = polytheism
    desc = pr_canaanite_choose_main_deity_decision_desc
    decision_tooltip = pr_canaanite_choose_main_deity_decision_tooltip

    is_shown = {
        # DLC check.
        has_fp1_dlc_trigger = yes
        # Following a valid religion & faith.
        OR = {
            faith = faith:semitic_polytheism
            faith = { has_doctrine_parameter = special_doctrine_semitic_patron_god_active }
        }
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:pr_canaanite_choose_main_deity_decision
            }
        }
    }

    is_valid = {
        piety >= 0
    }

    cost = {
        # Kosten via scripted values (pr_decision_values.txt)
        piety = @pr_choose_deity_piety_cost
    }

    cooldown = { years = 2 }

    effect = {
        add_to_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:pr_canaanite_choose_main_deity_decision
        }
        # Show the possible options.
        show_as_tooltip = {
            random_list = {
                desc = pr_canaanite_choose_main_deity_decision_tt
                100 = {
                    show_chance = no
                    desc = pr_canaanite_choose_main_deity_decision.select_ba_al_hadad
                    add_character_modifier = semitic_ba_al_hadad
                }
                100 = {
                    show_chance = no
                    desc = pr_canaanite_choose_main_deity_decision.select_ashera
                    add_character_modifier = semitic_ashera
                }
                100 = {
                    show_chance = no
                    desc = pr_canaanite_choose_main_deity_decision.select_dagan
                    add_character_modifier = semitic_dagan
                }
                100 = {
                    show_chance = no
                    desc = pr_canaanite_choose_main_deity_decision.select_el_elyon
                    add_character_modifier = semitic_el_elyon
                }
                100 = {
                    show_chance = no
                    desc = pr_canaanite_choose_main_deity_decision.select_ashtart
                    add_character_modifier = semitic_ashtart
                }
            }
        }
        trigger_event = pr_canaanite.0100
    }
    ai_check_interval = 96

    ai_potential = {
        piety >= minor_piety_value
        NOR = {
            has_character_modifier = semitic_ba_al_hadad
            has_character_modifier = semitic_ashera
            has_character_modifier = semitic_dagan
            has_character_modifier = semitic_el_elyon
            has_character_modifier = semitic_ashtart
        }
    }

    ai_will_do = {
        base = 100
    }
}

            has_title = title:k_carthago

        # Kosten via scripted values (pr_decision_values.txt)

        always = yes
    }

    ai_will_do = {
        base = 100
    }

    effect = {
        add_to_global_variable_list = {
            name = unavailable_unique_decisions

    }
