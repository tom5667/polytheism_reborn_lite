# --- Create Kingdom of Lower Egypt Decision ---
pr_egyptian_kingdom_of_lower_egypt_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/Memphis.dds"
    }
    desc = pr_egyptian_kingdom_of_lower_egypt_decision_desc
    decision_tooltip = pr_egyptian_kingdom_of_lower_egypt_decision_tooltip
    decision_group_type = major

    ai_check_interval = 4

    is_shown = {
        is_ruler = yes
        is_landed = yes
        OR = {
            has_game_rule = pr_ancient_kingdoms_decisions_on
            AND = {
                has_game_rule = pr_ancient_kingdoms_decisions_player
                is_ai = no
            }
        }
        OR = {
            religion = religion:egyptian_religion
            faith = faith:alexandrian_cult
            faith = faith:graeco_egyptian_polytheism
            culture = culture:kemetic
        }
        is_imprisoned = no
        is_at_war = no
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:pr_egyptian_kingdom_of_lower_egypt_decision
            }
        }
    }

    is_valid = {
        completely_controls = title:d_cairo
        completely_controls = title:d_sinai
        completely_controls = title:d_delta
        completely_controls = title:d_alexandria
        OR = {
            religion = religion:egyptian_religion
            culture = culture:kemetic
            faith = faith:alexandrian_cult
            faith = faith:graeco_egyptian_polytheism
        }
        is_imprisoned = no
        is_at_war = no
    }

    cost = {
        gold = pr_kingdom_restore_gold_cost
        prestige = pr_kingdom_restore_prestige_cost
    }

    effect = {
        add_to_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:pr_egyptian_kingdom_of_lower_egypt_decision
        }
        custom_tooltip = pr_egyptian_kingdom_of_lower_egypt_decision_effect_tooltip
        title:d_cairo = { save_scope_as = cairo }
        title:d_sinai = { save_scope_as = sinai }
        title:d_delta = { save_scope_as = delta }
        title:d_alexandria = { save_scope_as = alexandria }
        create_title_and_vassal_change = {
            type = created
            save_scope_as = title_change
            add_claim_on_loss = yes
        }
        title:k_lower_egypt = {
            change_title_holder = {
                holder = root
                change = scope:title_change
            }
        }
        resolve_title_and_vassal_change = scope:title_change
        title:d_cairo = { set_de_jure_liege_title = title:k_lower_egypt }
        title:d_sinai = { set_de_jure_liege_title = title:k_lower_egypt }
        title:d_delta = { set_de_jure_liege_title = title:k_lower_egypt }
        title:d_alexandria = { set_de_jure_liege_title = title:k_lower_egypt }
        hidden_effect = {
            title:k_lower_egypt = { set_de_jure_liege_title = title:e_egyptus }
            trigger_event = pr_egyptian_decision.0007
        }
    }

    ai_potential = {
        is_ruler = yes
        completely_controls = title:d_cairo
        completely_controls = title:d_delta
    }

    ai_will_do = {
        base = 100
        modifier = { add = 20 gold >= pr_kingdom_restore_gold_cost }
        modifier = { add = 20 prestige >= pr_kingdom_restore_prestige_cost }
    }
}

# --- Choose Main Deity (Kemetic) Decision ---
pr_egyptian_choose_main_deity_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/egyptian_temple.dds"
    }
    desc = pr_egyptian_choose_main_deity_decision_desc
    decision_tooltip = pr_egyptian_choose_main_deity_decision_tooltip
    decision_group_type = polytheism

    is_shown = {
        OR = {
            faith = faith:kemetic_polytheism
            faith = { has_doctrine_parameter = special_doctrine_kemetic_patron_god_active }
        }
        is_adult = yes
    }

    is_valid = {
    }

    is_valid_showing_failures_only = {
        piety >= 0
    }

    cost = {
        piety = 150
    }

    cooldown = { years = 2 }

    effect = {
        show_as_tooltip = {
            random_list = {
                desc = pr_egyptian_choose_main_deity_decision_tt
                100 = {
                    show_chance = no
                    desc = select_main_deity_kemetic_decision.select_amun_ra
                    add_trait = pr_kemetic_amun_ra
                }
                100 = {
                    show_chance = no
                    desc = select_main_deity_kemetic_decision.select_ra
                    add_trait = pr_kemetic_ra
                }
                100 = {
                    show_chance = no
                    desc = select_main_deity_kemetic_decision.select_isis
                    add_trait = pr_kemetic_isis
                }
                100 = {
                    show_chance = no
                    desc = select_main_deity_kemetic_decision.select_montu
                    add_trait = pr_kemetic_montu
                }
                100 = {
                    show_chance = no
                    desc = select_main_deity_kemetic_decision.select_hathor
                    add_trait = pr_kemetic_hathor
                }
            }
        }
        trigger_event = pr_egyptian_decision.0001
    }

    ai_check_interval = 90

    ai_potential = {
        piety >= minor_piety_value
        NOR = {
            has_trait = pr_kemetic_amun_ra
            has_trait = pr_kemetic_ra
            has_trait = pr_kemetic_isis
            has_trait = pr_kemetic_montu
            has_trait = pr_kemetic_hathor
        }
    }

    ai_will_do = {
        base = 100
    }
}
