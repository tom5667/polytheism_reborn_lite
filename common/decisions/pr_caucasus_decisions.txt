pr_caucasus_choose_main_deity_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/fp1_decisons_hold_bolt.dds"
    }
    decision_group_type = polytheism
    desc = pr_caucasus_choose_main_deity_decision_desc
    decision_tooltip = pr_caucasus_choose_main_deity_decision_tooltip

    is_shown = {
        # DLC check.
        has_fp1_dlc_trigger = yes
        # Following a valid religion & faith.
        religion = religion:caucasus_religion
        NOT = {
            is_target_in_global_variable_list = {
                name = unavailable_unique_decisions
                target = flag:pr_caucasus_choose_main_deity_decision
            }
        }
    }

    is_valid = {
        piety >= 0
    }

    cost = {
        # Kosten via scripted values (pr_decision_values.txt)
        piety = @pr_choose_deity_piety_cost
    }

    cooldown = { years = 2 }

    effect = {
        add_to_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:pr_caucasus_choose_main_deity_decision
        }
        # Show the possible options for Georgian Polytheism
        if = {
            limit = {
                faith = faith:georgian_polytheism
            }
            show_as_tooltip = {
                random_list = {
                    desc = pr_caucasus_choose_main_deity_decision_tt
                    100 = {
                        show_chance = no
                        desc = pr_caucasus_choose_main_deity_decision.select_armzi
                        add_character_modifier = georgian_armzi
                    }
                    100 = {
                        show_chance = no
                        desc = pr_caucasus_choose_main_deity_decision.select_adgilis_deda
                        add_character_modifier = georgian_adgilis_deda
                    }
                    100 = {
                        show_chance = no
                        desc = pr_caucasus_choose_main_deity_decision.select_kopala
                        add_character_modifier = georgian_kopala
                    }
                    100 = {
                        show_chance = no
                        desc = pr_caucasus_choose_main_deity_decision.select_barbale
                        add_character_modifier = georgian_the_barbale
                    }
                }
            }
            trigger_event = pr_caucasus_decision.0001
        }
        # Show the possible options for Armenian Polytheism
        else_if = {
            limit = {
                faith = faith:armenian_polytheism
            }
            show_as_tooltip = {
                random_list = {
                    desc = pr_caucasus_choose_main_deity_decision_tt
                    100 = {
                        show_chance = no
                        desc = pr_caucasus_choose_main_deity_decision.select_aramazd
                        add_character_modifier = armenian_aramazd
                    }
                    100 = {
                        show_chance = no
                        desc = pr_caucasus_choose_main_deity_decision.select_astlik
                        add_character_modifier = armenian_astlik
                    }
                    100 = {
                        show_chance = no
                        desc = pr_caucasus_choose_main_deity_decision.select_nane
                        add_character_modifier = armenian_nane
                    }
                    100 = {
                        show_chance = no
                        desc = pr_caucasus_choose_main_deity_decision.select_nvard
                        add_character_modifier = armenian_nvard
                    }
                }
            }
            trigger_event = pr_caucasus_decision.0002
        }
    }
    ai_check_interval = 96

    ai_potential = {
        piety >= minor_piety_value
        NOR = {
            has_character_modifier = georgian_armzi
            has_character_modifier = georgian_adgilis_deda
            has_character_modifier = georgian_kopala
            has_character_modifier = georgian_the_barbale
            has_character_modifier = armenian_aramazd
            has_character_modifier = armenian_astlik
            has_character_modifier = armenian_nane
            has_character_modifier = armenian_nvard
        }
    }

    ai_will_do = {
        base = 100
    }
}
