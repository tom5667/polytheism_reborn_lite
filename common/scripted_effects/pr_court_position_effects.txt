# =============================================
# SECTION 1: GENERIC APPOINT EFFECT
# =============================================

# Generic court position assignment effect
# Replaces 12 identical set_court_*_effect blocks
# Parameters:
#   $CANDIDATE$ — character to assign to the position
#   $POSITION$  — court position type (e.g. pr_court_augur_court_position)
#   $EMPLOYER$  — the court owner (liege)
# Scope: works from any scope, all scopes passed via parameters
pr_appoint_court_position_effect = {
	$CANDIDATE$ = {
		if = {
			limit = {
				NOR = {
					is_vassal_of = $EMPLOYER$
					is_courtier_of = $EMPLOYER$
					employer = $EMPLOYER$
				}
			}
			$EMPLOYER$ = { add_courtier = $CANDIDATE$ }
		}
	}
	$EMPLOYER$ = {
		if = {
			limit = {
				employs_court_position = $POSITION$
				any_court_position_holder = {
					type = $POSITION$
					count > 0
				}
			}
			every_court_position_holder = {
				type = $POSITION$
				save_temporary_scope_as = old_position_holder

				$EMPLOYER$ = {
					replace_court_position = {
						recipient = $CANDIDATE$
						holder = scope:old_position_holder
						court_position = $POSITION$
					}
					reverse_add_opinion = {
						target = scope:old_position_holder
						modifier = disappointed_opinion
						opinion = -15
					}
					reverse_add_opinion = {
						target = $CANDIDATE$
						modifier = hired_me_opinion
					}
				}
			}
		}
		else = {
			appoint_court_position = {
				recipient = $CANDIDATE$
				court_position = $POSITION$
			}
			reverse_add_opinion = {
				target = $CANDIDATE$
				modifier = hired_me_opinion
			}
		}
	}
}

# =============================================
# SECTION 3: OLD HOLDER EFFECT
# =============================================

# Saves the current holder of a court position as scope:old_holder
# Parameters:
#   $POS$      — court position type (e.g. pr_court_augur_court_position)
#   $EMPLOYER$ — the court owner
# Scope: works from any scope
pr_court_position_old_holder_effect = {
	$EMPLOYER$ = {
		if = {
			limit = {
				employs_court_position = $POS$
				any_court_position_holder = {
					type = $POS$
					count > 0
				}
			}
			every_court_position_holder = {
				type = $POS$
				limit = { is_physically_able = yes }
				save_scope_as = old_holder
			}
		}
	}
}
