# Roman Decision Scripted Effects
pr_romano_british_sr_notify_players_effect = {
	every_player = {
		limit = {
			Not = {this = root}
			capital_province = {
			geographical_region = world_europe_south_east
			}
		}
		trigger_event = pr_roman.0117
	}
}
 select_or_create_cult_leader_mithras_effect = {
	root = {
		every_vassal_or_below = {
			limit = {
				AND = {
					has_trait = mithras_cult
					has_trait = pr_pater_mithras
					religion = religion:roman_religion
				}
			}
		}
		save_scope_as = cult_leader_mithras
	}
	root = {
		every_courtier = {
			limit = {
				AND = {
					has_trait = mithras_cult
					has_trait = pr_pater_mithras
					religion = religion:roman_religion
				}
			}
			save_scope_as = cult_leader_mithras
		}
	}
	liege = {
		every_vassal_or_below = {

select_valid_daughter_and_save_scope = {

	# Loop through children and save up to 4 valid daughters
	root = {
		every_living_character = {
			limit = {
				any_parent = { this = root }
				#count > 0
				is_female = yes
				religion = religion:roman_religion
				age < 10
				NOT= { has_trait = vestalis }
			}
			save_scope_as = daughter
		}
	}

}
choose_new_vestalis_maxima = {
	random_living_character = {
		limit = {
			has_trait = vestalis
			age = { 30 41 }

		}
		save_scope_as =  new_vestalis_maxima
	}
	if = {
		limit = {
			NOT = {
				exists = scope:new_vestalis_maxima
			}
		}
		create_character = {
			employer = root
			random_traits = no
			gender = female
			age = {30 40}
			save_scope_as = new_vestalis_maxima
			trait = vestalis_maxima
			trait = vestalis
		}
	}
}
