# Scope: character. Clears the research chain state (pr_events.txt pr.0018-pr.0049) of this character, so a
# new attempt of pr_return_polytheism_decision starts clean. The chain keeps its state per character (not
# global), so several players can run it at the same time in multiplayer.
pr_research_ancient_religion_reset_effect = {
	if = { limit = { has_variable = research_great_library } remove_variable = research_great_library }
	if = { limit = { has_variable = research_small_library } remove_variable = research_small_library }
	if = { limit = { has_variable = learning_good } remove_variable = learning_good }
	if = { limit = { has_variable = bought_book } remove_variable = bought_book }
	clear_variable_list = research_anglo_saxon
	clear_variable_list = research_armenian
	clear_variable_list = research_assyrian
	clear_variable_list = research_atenism
	clear_variable_list = research_babylonian
	clear_variable_list = research_brythonic
	clear_variable_list = research_celtic_iberian
	clear_variable_list = research_dacian
	clear_variable_list = research_etruscan
	clear_variable_list = research_gaelic
	clear_variable_list = research_gaulish
	clear_variable_list = research_georgian
	clear_variable_list = research_german
	clear_variable_list = research_graeco_egyptian
	clear_variable_list = research_greek
	clear_variable_list = research_hittite
	clear_variable_list = research_illyrian
	clear_variable_list = research_kemetic
	clear_variable_list = research_latvian
	clear_variable_list = research_lithuanian
	clear_variable_list = research_lusitanian
	clear_variable_list = research_mycenaean
	clear_variable_list = research_prussian
	clear_variable_list = research_punic
	clear_variable_list = research_roman
	clear_variable_list = research_scythian
	clear_variable_list = research_semitic
	clear_variable_list = research_sumerian
	clear_variable_list = research_zalmoxism
}

pr_research_ancient_religion_events_great_library_effect = {
	if = {
		limit = {
			has_variable = research_great_library
		}
		trigger_event = {
			on_action = pr_research_ancient_religion_events_great_library
		}
	}
}

}

pr_research_ancient_religion_events_debate_effect = {
	if = {
		limit = {
			has_variable = learning_good
		}
		trigger_event = {
			on_action = pr_research_ancient_religion_events_debate
		}
	}
	if = {
		limit = {
			NOT = {
				has_variable = learning_good
			}
		}
		trigger_event = {
			on_action = pr_research_ancient_religion_events_debate_moderate_bad
		}
	}
}

pr_augur_auspice_effect = {
	if = {
		limit = {
			OR = {
				var:pr_augur_auspice = flag:pr_sacred_hens
				var:pr_augur_auspice = flag:pr_eagle
				var:pr_augur_auspice = flag:pr_crows
			}
		}
		trigger_event = {
			on_action = pr_augur_auspice_event
		}
	}
}

# Save court Augur as scope
save_court_augur_as_effect = {
	hidden_effect = {
		if = {
			limit = { exists = court_owner }
			court_owner = {
				random_court_position_holder = {
					type = pr_court_augur_court_position
					limit = {
						is_physically_able = yes
					}
					save_scope_as = $SCOPE_NAME$
				}
			}
		}
	}
}

# Section 7: Envoy Fate (Zoroastrian)
# =============================================

decide_fate_off_envoy = {
	# One outcome by the envoy's learning (the old ranges overlapped at 10 and 15 and skipped > 20;
	# the old IDs zoroastrianism.005-007 never existed, #146)
	if = {
		limit = { scope:envoy = { learning >= 15 } }
		trigger_event = pr_zoroastrian.0005
	}
	else_if = {
		limit = { scope:envoy = { learning >= 10 } }
		trigger_event = pr_zoroastrian.0006
	}
	else = {
		trigger_event = pr_zoroastrian.0007
	}
}
