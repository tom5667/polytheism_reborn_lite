	icon = pious_intent

	auto_complete = yes

	is_shown = {

	}

	on_invalidated = {
		trigger_event = activity_system.0090
	}

	ai_will_do = {

		if = {
			limit = { has_trait = zealous }

			multiply = 0.5
		}
	}

		animation = personality_zealous
	}
}
