#Visit temple of Zeus
pr_greek.0007 = {
	type = character_event
	title = pr_greek.0007.t
	desc = pr_greek.0007.desc
	theme = learning_scholarship_focus
	override_background = { reference = pr_temple_jupiter }
	left_portrait = {
		character = root
		animation = personality_content
	}

	option = {
		name =pr_greek.0007.a
		hidden_effect = {
			random_list = {
				30 = {
					send_interface_toast = {
						type = event_toast_effect_good
						left_icon = ROOT
						title = pr_greek.0007.message.positive
						add_character_modifier = {
							modifier = favoured_by_zeus
							years = 2
						}
					}
				}
				30 = {
					send_interface_toast = {
						type = event_toast_effect_good
						left_icon = ROOT
						title = pr_greek.0007.message.negative
						add_character_modifier = {
							modifier = unfavoured_by_zeus
							years = 2
						}
					}
				}
			}
		}
	}
}

#Buy book about Greek Mythology
pr_greek.0008= {
	trigger = {
		NOT = {
			any_character_artifact = {
				has_artifact_modifier = pr_book_about_mythology
			}
		}
	}
	type = character_event
	title = pr_greek.0008.t
	desc = pr_greek.0008.desc
	theme = learning_scholarship_focus
	override_background = { reference = market_west }
	left_portrait = {
		character = root
		animation = personality_content
	}

	option = {
		name = pr_greek.0008.a
		remove_short_term_gold = 15
		create_artifact = {
			name = greek_mythology_book
			description = greek_mythology_book_description
			type = miscellaneous
			visuals = book
			modifier = pr_book_about_mythology
		}

	}
}
