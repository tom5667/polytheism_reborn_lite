		animation = personality_zealous
	}

		character = scope:new_priest
		animation = happiness
	}

	immediate = {
		play_music_cue = "mx_cue_faith_conversion"
		save_scope_as = scoped_ruler
		if = { # Check if Temporal first.
			limit = {
				faith = { has_doctrine = doctrine_temporal_head }
			}
			save_scope_as = new_priest
		}
		else_if = { # Pick your chaplain first.
			limit = {
				exists = cp:councillor_court_chaplain
				cp:councillor_court_chaplain = { faith = scope:scoped_ruler.faith }
			}
			cp:councillor_court_chaplain = { save_scope_as = new_priest }
		}
		else = {
			hidden_effect = {
				create_character = {
					employer = root
					template = religious_leader_character
					random_traits = no
					save_scope_as = new_priest
					dynasty = generate
				}
			}
			scope:new_priest = {
				add_trait = education_learning_3
				add_trait = zealous
				add_trait = intellect_good_1
				random_list = {
					1 = { add_trait = lustful }
					1 = { add_trait = chaste }
					1 = { add_trait = wrathful }

pr_roman.0014 = {
	type = character_event
	title = pr_roman.0014.t
	desc =  pr_roman.0014.decision.desc
	theme = learning_scholarship_focus
	left_portrait = scope:high_skill_option
	lower_left_portrait = scope:mystic_option
	right_portrait = scope:low_skill_option
	lower_right_portrait = scope:excellent_skill_option

	trigger = { exists = capital_province } #If I lost it before I got this event, I shouldn't get a physician


	immediate = {
		hidden_effect = {
			#Find some appropriate options

			if = { #Performance heavy option finding is only for players
				limit = { is_ai = no }

				#EXCELLENT SKILL CHARACTER (for learned rulers)
				if = {
					limit = {
						OR = {
							has_learning_lifestyle_trait_trigger = yes
							learning >= high_skill_rating
						}
					}
					random_pool_character = {
						province = capital_province
						limit = {
							pr_excellent_skill_augur_trigger = yes
							has_trait = education_learning_3
						}
						alternative_limit = {
							pr_excellent_skill_augur_trigger = yes
						}
						weight = {
							base = 1
							modifier = {
								faith = root.faith
								factor = 10
							}
						}

						save_scope_as = excellent_skill_option
						if = {
							limit = { NOT = { has_trait = scholar } }
							add_trait = education_learning_3
							faith = root.faith
						}

						#Backup generation
						if = {
							limit = { NOT = { exists = scope:excellent_skill_option } }

							create_character = {
								location = root.capital_province
								gender_female_chance = 50
								template_character = root
								random_traits = yes
								add_trait = education_learning_3
								age = { 18 50 }
								faith = root.faith
								learning = high_skill_rating
								save_scope_as = excellent_skill_option
							}
							scope:excellent_skill_option = {
								add_trait = education_learning_3
							}
						}
					}
				}

				#HIGH-SKILL CHARACTER
				random_pool_character = {
					province = capital_province
					limit = {
						pr_high_skill_augur_trigger = yes
						trigger_if = {
							limit = {
								exists = scope:excellent_skill_option
							}
							NOT = { this = scope:excellent_skill_option }
						}
					}
					weight = {
						base = 1
						modifier = {
							has_trait = lifestyle_mystic
							factor = 100
						}
						modifier = {
							faith = root.faith
							factor = 10
						}
					}
					save_scope_as = high_skill_option
				}

				#Backup generation
				if = {
					limit = { NOT = { exists = scope:high_skill_option } }

					create_character = {
						location = root.capital_province
						gender_female_chance = 50
						template_character = root
						random_traits = yes
						add_trait = education_learning_2
						age = { 18 50 }
						faith = root.faith
						learning = { medium_skill_rating high_skill_rating }
						save_scope_as = high_skill_option
					}
				}

				#Set augur trait if skill is too low
				if = {
					limit = { exists = scope:high_skill_option }
					scope:high_skill_option = {
						set_to_lowborn = yes
						while = {
							limit = {
								OR = {
								AND = {
										root = { highest_held_title_tier >= tier_kingdom }
										learning < high_skill_rating
									}
									AND = {
										root = { highest_held_title_tier <= tier_duchy }
										learning < medium_skill_rating
									}
								}
								NOT = { has_trait = education_learning_3 }
							}
							mystic_lifestyle_rank_up_effect = yes
						}
					}
				}

				#MYSTIC CHARACTER (optional, only if available in pool)
				random_pool_character = {
					province = root.capital_province
					limit = {
						pr_mystic_augur_trigger = yes
						trigger_if = {
							limit = {
								exists = scope:excellent_skill_option
							}
							NOT = { this = scope:excellent_skill_option }
						}
						trigger_if = {
							limit = {
								exists = scope:high_skill_option
							}
							NOT = { this = scope:high_skill_option }
						}
					}
					weight = {
						base = 1
						modifier = {
							has_trait = lifestyle_mystic
							factor = 100
						}
						modifier = {
							faith = root.faith
							factor = 10
						}
					}
					save_scope_as = mystic_option
				}

				#Set mystic trait if skill is too low
				if = {
					limit = { exists = scope:mystic_option }
					scope:mystic_option = {
						while = {
							limit = {
								OR = {
									AND = {
										root = { highest_held_title_tier >= tier_kingdom }
										learning < high_skill_rating
									}
									AND = {
										root = { highest_held_title_tier <= tier_duchy }
										learning < medium_skill_rating
									}
								}
								NOT = { has_trait = mystic_3 }
							}
							ai_mystic_lifestyle_rank_up_effect = yes
						}
					}
				}

				#Low-medium skill character
				random_pool_character = {
					province = root.capital_province
					limit = {
						pr_low_skill_augur_trigger = yes
						trigger_if = {
							limit = {
								exists = scope:excellent_skill_option
							}
							NOT = { this = scope:excellent_skill_option }
						}
						trigger_if = {
							limit = {
								exists = scope:high_skill_option
							}
							NOT = { this = scope:high_skill_option }
						}
						trigger_if = {
							limit = {
								exists = scope:mystic_option
							}
							NOT = { this = scope:mystic_option }
						}
					}
					weight = {
						base = 1
						modifier = {
							faith = root.faith
							factor = 10
						}
					}
					save_scope_as = low_skill_option
				}
			}
			else = { #Ai get someone simple
				random_pool_character = {
					province = root.capital_province
					limit = { pr_low_skill_augur_light_trigger = yes }
					weight = {

#--- The Greek Opposition (0050) ---
pr_roman.0050 = {
	type = character_event
	desc = pr_roman.0050.desc
	title = pr_roman.0050.t
	theme = learning_scholarship_focus

	right_portrait = {
		character = root
		animation = personality_content
	}
	left_portrait = {
		character = scope:pr_greek_opponent
		animation = argumentative
	}
	override_background = { reference = throne_room }

	option = {
		name = pr_roman.0050.a # Greek and Latin shall coexist

		scope:pr_greek_opponent = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 15
			}
		}
		add_prestige = 25
		# A small learning boost from the cultural exchange
		add_character_modifier = {
			modifier = pr_latin_revival_diplomatic
			years = 5
		}
		# The Latin speaker appreciates the diplomatic approach
		if = {
			limit = { is_alive = scope:pr_latin_speaker }
			scope:pr_latin_speaker = {
				add_opinion = {
					target = root
					modifier = grateful_opinion
					opinion = 10
				}
			}
			trigger_event = {
				days = 14
				id = pr_roman.0051
			}
		}
	}

	option = {
		name = pr_roman.0050.b # Latin is the voice of Rome

		subtract_prestige = 50
		scope:pr_greek_opponent = {
			add_opinion = {
				target = root
				modifier = disrespect_opinion
				opinion = -15
			}
		}
		# The Latin speaker feels emboldened
		if = {
			limit = { is_alive = scope:pr_latin_speaker }
			scope:pr_latin_speaker = {
				add_trait_xp = {
					trait = lifestyle_scholar
					value = { 50 100 }
				}
			}
			trigger_event = {
				days = 14
				id = pr_roman.0051
			}
		}
	}

	option = {
		name = pr_roman.0050.c # Perhaps you should learn Latin yourself

		scope:pr_greek_opponent = {
			learn_language_of_culture = culture:roman
			random = {
				chance = 50
				add_trait = stubborn
			}
			add_opinion = {
				target = root
				modifier = insult_opinion
				opinion = -25
			}
		}
		add_stress = minor_stress_impact_gain

		# Schedule the scriptory event
		if = {
			limit = { is_alive = scope:pr_latin_speaker }
			trigger_event = {
				days = 14
				id = pr_roman.0051
			}
		}
	}
}

#--- The Scriptory (0051) ---
pr_roman.0051 = {
	type = character_event
	desc = pr_roman.0051.desc
	title = pr_roman.0051.t
	theme = learning_scholarship_focus

	right_portrait = {
		character = root
		animation = personality_content
	}
	left_portrait = {
		character = scope:pr_latin_speaker
		animation = storyteller
	}
	override_background = { reference = throne_room }

	option = {
		name = pr_roman.0051.a # Establish the scriptorium
		trigger = { gold >= 150 }

		subtract_gold = 150
		scope:pr_latin_speaker = {
			add_trait = lifestyle_scholar
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}
		# Courtiers gradually learn Latin
		any_courtier = {
			limit = {
				NOT = { knows_language_of_culture = culture:roman }
				NOT = { is_landed = yes }
			}
			learn_language_of_culture = culture:roman
		}
		add_character_modifier = {
			modifier = pr_scriptorium_established
			years = 10
		}
	}

	option = {
		name = pr_roman.0051.b # We cannot afford such luxury
		fallback = yes

		scope:pr_latin_speaker = {
			add_opinion = {
				target = root
				modifier = disrespect_opinion
				opinion = -10
			}
		}
		add_stress = minor_stress_impact_gain
	}

	option = {
		name = pr_roman.0051.c # Make it a grand academy!
		trigger = { gold >= 300 }

		subtract_gold = 300
		scope:pr_latin_speaker = {
			add_trait = lifestyle_scholar
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 30
			}
			random = {
				chance = 33
				add_trait = shrewd
			}
		}
		# All courtiers learn Latin
		any_courtier = {
			limit = {
				NOT = { knows_language_of_culture = culture:roman }
			}
			learn_language_of_culture = culture:roman
		}
		add_character_modifier = {
			modifier = pr_grand_academy_established
			years = 15
		}
		add_prestige = 100
	}
}
