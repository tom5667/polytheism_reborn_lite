#Visiting Temple of Tanit
pr_canaanite.0005 = {
	type = character_event
	title = pr_canaanite.0005.t
	desc = pr_canaanite.0005.desc
	theme = learning
	override_background = { reference = temple_generic }
	immediate = {
		hidden_effect={
			cp:councillor_court_chaplain = {
				save_scope_as = tanit_priest
			}
		}
	}
	right_portrait = {
		character = root
		animation = personality_content
	}
	left_portrait = {
		character = scope:tanit_priest
		animation = personality_content
	}
	option = {
		name = pr_canaanite.0005.a
		hidden_effect = {
			random_list = {
				30 = {
					send_interface_toast = {
						type = event_toast_effect_good
						left_icon = ROOT
						title = pr_canaanite.0005.message.positive
						add_character_modifier = {
							modifier = favoured_by_tanit
							years = 2
						}
					}
				}
				30 = {
					send_interface_toast = {
						type = event_toast_effect_good
						left_icon = ROOT
						title = pr_canaanite.0005.message.negative
						add_character_modifier = {
							modifier = unfavoured_by_tanit
							years = 2
						}
					}
				}
			}
		}	
	}
}
