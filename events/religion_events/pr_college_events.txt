# College of Pontiffs (#76). Storyline "The New Believers", towards the Lex Ogulnia (0100-0199).
# Design: chronicle/decisions_log.md (2026-10-20, 2026-10-22); sources: resource/college_laws_research.md.
# 0100 the petition, 0101 the Church reaches for the converts, 0102 the vote, 0103 the law passes, 0104 the veto,
# 0105 the Church answers (after the envoy was sent home).
# Started by pr_college_ogulnia_yearly_effect (yearly pulse) or the debug decision; 0102 again through
# pr_college_bring_lex_ogulnia_decision after a veto.
# Decided 2026-10-22: every choice costs or yields something real. The vote chance comes from
# var:pr_college_ogulnia_support (set in 0101) and the refusal count, not from opinions.

# The petition
pr_college.0100 = {
	type = character_event
	title = pr_college.0100.t
	desc = {
		desc = pr_college.0100.desc
		first_valid = {
			triggered_desc = {
				trigger = { faith:roman_polytheism = { has_variable = pr_college_ogulnia_refusals } }
				desc = pr_college.0100.desc.again
			}
			desc = pr_college.0100.desc.first
		}
		first_valid = {
			triggered_desc = {
				trigger = { current_date < 800.1.1 }
				desc = pr_college.era.700
			}
			triggered_desc = {
				trigger = { current_date < 1050.1.1 }
				desc = pr_college.era.867
			}
			triggered_desc = {
				trigger = { current_date < 1150.1.1 }
				desc = pr_college.era.1066
			}
			desc = pr_college.era.1178
		}
	}
	theme = faith
	override_background = { reference = pr_temple } # not vanilla temple: it can show a church (Hagia Sophia for an emperor in Constantinople)
	left_portrait = {
		character = scope:pr_ogulnia_convert
		animation = personality_zealous
	}
	right_portrait = {
		character = scope:pr_ogulnia_founder
		animation = disapproval
	}

	immediate = {
		pr_college_ogulnia_setup_scopes_effect = yes
		# A new round starts clean: support and vetoes from an earlier round do not carry over
		if = { limit = { has_variable = pr_college_ogulnia_support } remove_variable = pr_college_ogulnia_support }
		if = { limit = { has_variable = pr_college_ogulnia_vetoes } remove_variable = pr_college_ogulnia_vetoes }
		if = { limit = { has_variable = pr_college_ogulnia_veto_cooldown } remove_variable = pr_college_ogulnia_veto_cooldown }
	}

	# Let the College hear the petition
	option = {
		name = pr_college.0100.a
		scope:pr_ogulnia_convert = {
			add_opinion = {
				target = root
				modifier = pr_college_convert_grateful_opinion
			}
		}
		trigger_event = {
			id = pr_college.0101
			days = { 20 40 }
		}
		ai_chance = {
			base = 60
			modifier = { add = 20 has_trait = just }
			modifier = { add = 20 has_trait = gregarious }
			modifier = { add = -20 has_trait = arrogant }
		}
	}

	# The College is not yours to ask for
	option = {
		name = pr_college.0100.b
		pr_college_ogulnia_refuse_effect = yes
		ai_chance = {
			base = 40
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 20 has_trait = arrogant }
		}
	}
}

# The Church reaches for the converts. Decided 2026-10-22, each choice has its own price:
# protect = gold, a better chance (+20); let the preacher talk = a gamble (holds: +30; relapse: the round ends);
# send the envoy home = free, a small bonus (+5), but the Church answers later (pr_college.0105). Second round after
# the review: protect 15 -> 20, holds 25 -> 30, envoy 10 -> 5, so protecting is no longer dominated by the envoy.
pr_college.0101 = {
	type = character_event
	title = pr_college.0101.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:pr_ogulnia_church_head }
				desc = pr_college.0101.desc.head
			}
			desc = pr_college.0101.desc.nohead
		}
		first_valid = {
			triggered_desc = {
				trigger = { current_date < 800.1.1 }
				desc = pr_college.0101.era.700
			}
			triggered_desc = {
				trigger = { current_date < 1050.1.1 }
				desc = pr_college.0101.era.867
			}
			triggered_desc = {
				trigger = { current_date < 1150.1.1 }
				desc = pr_college.0101.era.1066
			}
			desc = pr_college.0101.era.1178
		}
	}
	theme = faith
	override_background = { reference = throne_room }
	left_portrait = {
		character = scope:pr_ogulnia_convert
		animation = worry
	}
	right_portrait = {
		trigger = { exists = scope:pr_ogulnia_church_head }
		character = scope:pr_ogulnia_church_head
		animation = personality_rational
	}

	trigger = {
		pr_college_ogulnia_ruler_still_valid_trigger = yes
	}
	# The ruler no longer qualifies: the round ends, so the storyline does not block other rulers
	on_trigger_fail = {
		pr_college_ogulnia_collapse_effect = yes
	}

	# The convert may have died since the petition: a new one steps forward
	immediate = {
		pr_college_ogulnia_setup_scopes_effect = yes
	}

	# They are under my protection: costs gold, the new believers rally, the founders are displeased
	option = {
		name = pr_college.0101.a
		trigger = { short_term_gold >= medium_gold_value }
		show_as_unavailable = { short_term_gold < medium_gold_value }
		remove_short_term_gold = medium_gold_value
		pr_college_ogulnia_founders_opinion_effect = { MODIFIER = pr_college_founders_displeased_opinion }
		custom_tooltip = pr_college.0101.a.tt
		hidden_effect = {
			pr_college_ogulnia_set_support_effect = { SUPPORT = 20 }
		}
		custom_tooltip = pr_college.next_vote.tt # when the next step comes (playtest 2026-11-07)
		trigger_event = {
			id = pr_college.0102
			days = { 30 60 }
		}
		ai_chance = {
			base = 40
			modifier = { add = 20 has_trait = generous }
			modifier = { add = 20 has_trait = just }
		}
	}

	# Let the preacher talk: the convert may fall back to the Church, and then the petition loses its face
	option = {
		name = pr_college.0101.b
		random_list = {
			60 = {
				desc = pr_college.0101.b.holds
				modifier = { add = 20 scope:pr_ogulnia_convert = { has_trait = zealous } }
				modifier = { add = -20 scope:pr_ogulnia_convert = { has_trait = cynical } }
				scope:pr_ogulnia_convert = { add_piety = minor_piety_gain }
				custom_tooltip = pr_college.0101.b.holds.tt
				hidden_effect = {
					pr_college_ogulnia_set_support_effect = { SUPPORT = 30 }
				}
				custom_tooltip = pr_college.next_vote.tt # when the next step comes (playtest 2026-11-07)
				trigger_event = {
					id = pr_college.0102
					days = { 30 60 }
				}
			}
			40 = {
				desc = pr_college.0101.b.lost
				scope:pr_ogulnia_convert = {
					set_character_faith = scope:pr_ogulnia_church
					set_character_rite = scope:pr_ogulnia_church_rite
				}
				pr_college_ogulnia_collapse_effect = yes
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = content }
			modifier = { add = 20 has_trait = brave }
		}
	}

	# Send the envoy home: the founders are pleased, the Church will answer
	option = {
		name = pr_college.0101.c
		pr_college_ogulnia_founders_opinion_effect = { MODIFIER = pr_college_founders_honoured_opinion }
		scope:pr_ogulnia_church_head ?= {
			add_opinion = {
				target = root
				modifier = pr_college_church_rebuffed_opinion
			}
		}
		custom_tooltip = pr_college.0101.c.tt
		hidden_effect = {
			pr_college_ogulnia_set_support_effect = { SUPPORT = 5 }
		}

			id = pr_college.0102
			days = { 30 60 }
		}
		trigger_event = {
			id = pr_college.0105
			days = { 365 1095 }
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 20 has_trait = wrathful }
		}
	}
}

# The vote on the Lex Ogulnia. Decided 2026-10-22: open = base 40, gradual = base 60. Support from the letters is added
# to the success side, every refusal so far (max 3) adds 10 to the failure side, every veto 15 to the success side.

	type = character_event
	title = pr_college.0102.t
	desc = {
		desc = pr_college.0102.desc
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = pr_college_ogulnia_vetoes }

		character = scope:pr_ogulnia_founder

	}
	right_portrait = {
		character = scope:pr_ogulnia_convert

	}

	trigger = {
		NOT = { exists = global_var:pr_college_lex_ogulnia }
		pr_college_ogulnia_ruler_still_valid_trigger = yes
	}
	# The ruler no longer qualifies: the round ends, so the storyline does not block other rulers
	on_trigger_fail = {
		pr_college_ogulnia_collapse_effect = yes
	}

	immediate = {
		pr_college_ogulnia_setup_scopes_effect = yes
	}

	# Open the College: the hardest to carry, the greatest gain
	option = {
		name = pr_college.0102.a
		skill = diplomacy
		custom_tooltip = pr_college.0102.a.tt
		duel = {
			skill = diplomacy
			target = scope:pr_ogulnia_founder
			40 = {
				desc = pr_college.0102.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				# Traits of the ruler (decided 2026-10-22: the worse, the smaller the chance)
				modifier = { add = 10 has_trait = zealous }
				modifier = { add = 10 has_trait = just }
				modifier = { add = 10 has_trait = honest }
				modifier = { add = 10 has_trait = gregarious }
				modifier = { add = 10 has_trait = diplomat }
				modifier = { add = -10 has_trait = arrogant }
				modifier = { add = -15 has_trait = cynical }
				modifier = { add = -10 has_trait = deceitful }
				modifier = { add = -10 has_trait = shy }
				modifier = { add = -10 has_trait = wrathful }
				# Support from the Church letters (pr_college.0101)
				modifier = { add = 5 var:pr_college_ogulnia_support ?= 5 }
				modifier = { add = 20 var:pr_college_ogulnia_support ?= 20 }
				modifier = { add = 30 var:pr_college_ogulnia_support ?= 30 }
				# Brought again after a veto (Livy 10.9: vetoed first, passed the next day)
				modifier = { add = 15 var:pr_college_ogulnia_vetoes ?= 1 }

				modifier = { add = 45 has_variable = pr_college_ogulnia_vetoes var:pr_college_ogulnia_vetoes >= 3 }
				min = 10
				save_scope_value_as = { name = pr_ogulnia_mode value = flag:open }
				pr_college_ogulnia_pass_effect = { OPEN = yes }
				trigger_event = pr_college.0103
			}
			60 = {
				desc = pr_college.0102.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				# Every earlier refusal embittered the new believers (decided 2026-10-22: -10 each, at most 3)
				modifier = { add = 10 faith:roman_polytheism = { var:pr_college_ogulnia_refusals ?= 1 } }
				modifier = { add = 20 faith:roman_polytheism = { var:pr_college_ogulnia_refusals ?= 2 } }
				modifier = { add = 30 faith:roman_polytheism = { has_variable = pr_college_ogulnia_refusals var:pr_college_ogulnia_refusals >= 3 } }
				min = 10
				pr_college_ogulnia_veto_effect = yes
				trigger_event = pr_college.0104
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = ambitious }
			modifier = { add = 20 has_trait = brave }
		}
	}

# The Church answers, a year or three after its envoy was sent home (pr_college.0101 c). Decided 2026-10-22: an event
# with a choice and opinion, no casus belli: make amends with gold or prestige (the Church head forgives the envoy),
# or defy the Church (piety, but the
# Christian rulers around you think less of you for 10 years).
pr_college.0105 = {
	type = character_event
	title = pr_college.0105.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:pr_ogulnia_church_head }
				desc = pr_college.0105.desc.head
			}
			desc = pr_college.0105.desc.nohead
		}
		first_valid = {
			triggered_desc = {
				trigger = { current_date < 800.1.1 }
				desc = pr_college.0105.era.700
			}
			triggered_desc = {
				trigger = { current_date < 1050.1.1 }
				desc = pr_college.0105.era.867
			}
			triggered_desc = {
				trigger = { current_date < 1150.1.1 }
				desc = pr_college.0105.era.1066
			}
			desc = pr_college.0105.era.1178
		}
	}
	theme = faith
	override_background = { reference = throne_room }
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		trigger = { exists = scope:pr_ogulnia_church_head }
		character = scope:pr_ogulnia_church_head
		animation = disapproval
	}

	trigger = {
		is_alive = yes
		is_landed = yes
		faith = faith:roman_polytheism
		NOT = { government_has_flag = government_is_theocracy }
	}

	# The head of the Church may have died in the meantime
	immediate = {
		save_scope_as = pr_ogulnia_ruler
		pr_college_ogulnia_find_church_effect = yes
	}

	# Send gifts to make amends
	option = {
		name = pr_college.0105.a
		trigger = { short_term_gold >= medium_gold_value }
		show_as_unavailable = { short_term_gold < medium_gold_value }
		remove_short_term_gold = medium_gold_value
		custom_tooltip = pr_college.0105.amends.tt
		scope:pr_ogulnia_church_head ?= {
			remove_opinion = {
				target = root
				modifier = pr_college_church_rebuffed_opinion
			}
		}
		# The quarrel ends here; the law goes on as before (playtest 2026-11-06: players asked "what now?")
		pr_college_church_answers_end_effect = { LAW = ogulnia }
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = craven }
			modifier = { factor = 0 short_term_gold < medium_gold_value }
		}
	}

	# Receive the envoy with honour this time
	option = {
		name = pr_college.0105.b
		add_prestige = medium_prestige_loss
		custom_tooltip = pr_college.0105.amends.tt
		scope:pr_ogulnia_church_head ?= {
			remove_opinion = {
				target = root
				modifier = pr_college_church_rebuffed_opinion
			}
		}
		# The quarrel ends here; the law goes on as before (playtest 2026-11-06: players asked "what now?")
		pr_college_church_answers_end_effect = { LAW = ogulnia }
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = humble }
		}
	}

	# The gods of Rome answer to no bishop
	option = {
		name = pr_college.0105.c
		add_piety = medium_piety_gain
		custom_tooltip = pr_college.0105.c.tt
		hidden_effect = {
			every_neighboring_and_across_water_top_liege_realm_owner = {
				limit = { is_christian_trigger = yes } # character scope: religion_tag is a faith trigger
				add_opinion = {
					target = root
					modifier = pr_college_church_defied_opinion
