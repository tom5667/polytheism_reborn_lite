#Visiting Temple of Jupiter
punic.005 = {
	type = character_event
	title = punic.005.t
	desc = punic.005.desc
	theme = learning
	override_background = { reference = temple_generic }
	immediate = {
		hidden_effect={
			cp:councillor_court_chaplain = {
				save_scope_as = tanit_priest
			}
		}
	}
	right_portrait = {
		character = root
		animation = personality_content
	}
	left_portrait = {
		character = scope:tanit_priest
		animation = personality_content
	}
	option = {
		name = punic.005.a
		hidden_effect = {
			random_list = {
				30 = {
					send_interface_toast = {
						type = event_toast_effect_good
						left_icon = ROOT
						title = punic.005.message.positive
						add_character_modifier = {
							modifier = favoured_by_tanit
							years = 2
						}
					}
				}
				30 = {
					send_interface_toast = {
						type = event_toast_effect_good
						left_icon = ROOT
						title = punic.005.message.negative
						add_character_modifier = {
							modifier = unfavoured_by_tanit
							years = 2
						}
					}
				}
			}
		}	
	}
}
