#Spend money to build a small local shrine
pr_daco_thracian.0009 = {
	type = character_event
	title = pr_daco_thracian.0009.t
	desc = pr_daco_thracian.0009.desc
	theme = learning
	override_background = { reference= pr_generic_temple }
	immediate = {
		hidden_effect = {
			create_character = {
				employer = root
				random_traits = yes
				dynasty = generate
				save_scope_as = random_character
				stewardship = 26
			}
			cp:councillor_court_chaplain = {
				save_scope_as = realm_priest
			}
		}
	}
	right_portrait = {
		character = root
		animation = personality_content
	}
	left_portrait = {
		character = scope:realm_priest
	}
	#pay for everything
	option = {
		name = pr_daco_thracian.0009.a
		add_character_modifier = {
			modifier = built_local_shrine
			years = 5
		}
		add_trait = zealous
		remove_short_term_gold = 200
		scope:realm_priest = {
			add_opinion = {
				modifier = funded_local_shrine_opinion
				target = root
			}
		}
	}
	#Pay only for the materials
	option = {
		name = pr_daco_thracian.0009.b
		remove_short_term_gold = 50
		scope:realm_priest = {
			add_opinion = {
				modifier = helped_fund_local_shrine_opinion
				target = root
			}
		}
	}
	#Don't help them and wish them well
	option = {
		name = pr_daco_thracian.0009.c
		scope:realm_priest = {
			add_opinion = {
				modifier = refused_to_fund_local_shrine_opinion
				target = root
			}
		}
	}
}


# --- Zalmoxis Human Sacrifice (0010) ---
pr_daco_thracian_decision.0010 = {
	type = character_event
	title = pr_daco_thracian_decision.0010.t
	desc = pr_daco_thracian_decision.0010.desc
	theme = realm
	override_background = { reference = wilderness_mountains }

	immediate = {
		random_dummy_gender_soldier_effect = { SCOPE_NAME = sacrifice }
		hidden_effect = {
			create_character = {
				age = { 25 40 }
				random_traits = yes
				prowess = { 1 10 }
				martial = { 9 15 }
				dynasty = none
				gender = male
				faith = root.faith
				culture = root.culture
				location = root.title_province
				save_scope_as = sacrifice
			}
			# Save them in the designated county.
			root.title_province = {
				set_variable = {
					name = fearless_warrior
					value = scope:sacrifice
					years = fp1_stele_duration_value
				}
			}
			cp:councillor_court_chaplain = {
				save_scope_as = realm_priest
			}
		}
	}

	right_portrait = {
		character = root
		animation = personality_content
	}
	left_portrait = {
		character = scope:sacrifice
		animation = celebrate_sword
	}

	# Choose the soldier
	option = {
		name = pr_daco_thracian_decision.0010.a
		trigger_event = {
			days = 5
			id = pr_daco_thracian_decision.0011
		}
	}

	# Stop the sacrifice
	option = {
		name = pr_daco_thracian_decision.0010.b
		add_character_modifier = {
			modifier = sacrifice_refused
			years = 1
		}
	}
}
