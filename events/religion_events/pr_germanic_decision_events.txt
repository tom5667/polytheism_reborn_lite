	left_portrait = {
		character = root

		animation = shock
	}
	weight_multiplier = {
		base = 1
		modifier = {
			add = 0.2
			has_trait = zealous
		}
		modifier = {
			add = -0.2
			has_trait = cynical
		}
		modifier = {
			add = 0.2
			has_trait = brave
		}
		modifier = {
			add = -0.2
			has_trait = craven
		}
		modifier = {
			add = 0.2
			has_trait = ambitious
		}
		modifier = {
			add = -0.2
			has_trait = content
		}
	}
	option = { # Just a bad dream
		name = pr_germanic.0005.a
		if = {
			limit = {
				OR = {
					has_trait = zealous
					has_trait = cynical
				}
			}
			stress_impact = {
				zealous = medium_stress_impact_gain
				cynical = miniscule_stress_impact_gain
			}
		}
		else = {
			add_stress = minor_stress_impact_gain
		}

	}
}
